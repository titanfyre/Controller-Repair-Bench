cmake_minimum_required(VERSION 3.20)
cmake_policy(SET CMP0091 NEW)
project(ControllerRepairBench VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Use Windows x64 and Visual Studio 2022 C++ tools.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(SDK_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/SDK" CACHE PATH "Matching demo or licensed production SDK directory")
add_compile_definitions(UNICODE _UNICODE NOMINMAX)
add_compile_options(/W4 /WX /utf-8 /permissive-)
add_library(InterceptorSDK STATIC "${SDK_DIRECTORY}/Interceptor.cpp")
target_include_directories(InterceptorSDK PUBLIC "${SDK_DIRECTORY}")
target_link_libraries(InterceptorSDK PUBLIC setupapi newdev cfgmgr32 hid advapi32 user32 comctl32 ole32)
add_executable(ControllerRepairBench WIN32 Source/Main.cpp)
target_link_libraries(ControllerRepairBench PRIVATE InterceptorSDK shell32 comdlg32 gdi32)
target_link_options(ControllerRepairBench PRIVATE "/MANIFESTUAC:level='asInvoker' uiAccess='false'" "/PDBALTPATH:%_PDB%")
set_target_properties(ControllerRepairBench PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/bin")
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
  set_target_properties(ControllerRepairBench PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_CURRENT_SOURCE_DIR}/bin")
endforeach()
enable_testing()
add_test(NAME OfflineModelFixtures COMMAND ControllerRepairBench --self-test)
